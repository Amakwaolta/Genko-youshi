<script>
'use strict';
const $=id=>document.getElementById(id);
const STORAGE_KEY='manuscript-editor-v2';
const DEFAULTS={direction:'horizontal',rows:20,cols:20,cellSize:28,gridColor:'#d5a7a7',font:'"Yu Mincho","Hiragino Mincho ProN","Noto Serif JP",serif',fontSize:18,title:'',bodyText:''};
let settings={...DEFAULTS};
function clamp(value,min,max){const n=Number.parseInt(value,10);return Number.isFinite(n)?Math.min(max,Math.max(min,n)):min;}
function readControls(){settings.rows=clamp($('rows').value,5,60);settings.cols=clamp($('cols').value,5,60);settings.cellSize=clamp($('cellSize').value,18,42);settings.fontSize=clamp($('fontSize').value,12,36);settings.gridColor=$('gridColor').value;settings.font=$('font').value;settings.title=$('title').value;settings.bodyText=$('bodyText').value;}
function render(){readControls();const p=$('paper'),cell=settings.cellSize;p.style.setProperty('--rows',settings.rows);p.style.setProperty('--cols',settings.cols);p.style.setProperty('--cell',cell+'px');p.style.setProperty('--grid-color',settings.gridColor);p.style.setProperty('--text-size',settings.fontSize+'px');p.style.width=(settings.cols*cell)+'px';p.style.height=(settings.rows*cell)+'px';p.className='paper '+settings.direction;$('bodyText').style.fontFamily=settings.font;$('cellLabel').textContent=cell+' px';$('fontLabel').textContent=settings.fontSize+' px';$('orientationLabel').textContent=(settings.direction==='vertical'?'縦書き':'横書き')+'プレビュー';$('horizontalBtn').classList.toggle('active',settings.direction==='horizontal');$('verticalBtn').classList.toggle('active',settings.direction==='vertical');$('horizontalBtn').setAttribute('aria-pressed',String(settings.direction==='horizontal'));$('verticalBtn').setAttribute('aria-pressed',String(settings.direction==='vertical'));$('dimensions').textContent=settings.rows+'行 × '+settings.cols+'列';$('count').textContent=settings.bodyText.length.toLocaleString('ja-JP')+'文字';$('capacity').textContent=(settings.rows*settings.cols).toLocaleString('ja-JP')+'マス';}
function saveState(){try{readControls();localStorage.setItem(STORAGE_KEY,JSON.stringify(settings));$('saveState').textContent='保存済み · '+new Date().toLocaleTimeString('ja-JP',{hour:'2-digit',minute:'2-digit'});}catch(error){$('saveState').textContent='自動保存できません。TXT保存でバックアップしてください。';}}
function updateAndSave(){render();saveState();}
function restore(){try{const stored=localStorage.getItem(STORAGE_KEY);if(!stored)return;const data=JSON.parse(stored);if(!data||typeof data!=='object')return;settings={...DEFAULTS,...data};settings.direction=settings.direction==='vertical'?'vertical':'horizontal';$('rows').value=clamp(settings.rows,5,60);$('cols').value=clamp(settings.cols,5,60);$('cellSize').value=clamp(settings.cellSize,18,42);$('fontSize').value=clamp(settings.fontSize,12,36);$('gridColor').value=/^#[0-9a-f]{6}$/i.test(settings.gridColor)?settings.gridColor:DEFAULTS.gridColor;$('font').value=settings.font;$('title').value=String(settings.title||'').slice(0,80);$('bodyText').value=String(settings.bodyText||'');}catch(error){$('saveState').textContent='保存データを読み込めませんでした。';}}
['rows','cols','cellSize','gridColor','font','fontSize','title'].forEach(id=>$(id).addEventListener('input',updateAndSave));$('bodyText').addEventListener('input',updateAndSave);
$('horizontalBtn').addEventListener('click',()=>{settings.direction='horizontal';updateAndSave()});$('verticalBtn').addEventListener('click',()=>{settings.direction='vertical';updateAndSave()});
$('pdfBtn').addEventListener('click',()=>{document.title=( $('title').value.trim()||'原稿用紙')+' - 原稿用紙エディタ';window.print()});
$('txtBtn').addEventListener('click',()=>{const blob=new Blob(['\ufeff',$('bodyText').value],{type:'text/plain;charset=utf-8'});const url=URL.createObjectURL(blob),a=document.createElement('a');a.href=url;a.download=( $('title').value.trim()||'原稿用紙').replace(/[\\/:*?"<>|]/g,'_')+'.txt';document.body.appendChild(a);a.click();a.remove();setTimeout(()=>URL.revokeObjectURL(url),1000);});
$('clearBtn').addEventListener('click',()=>{if($('bodyText').value&&confirm('本文をすべて消去します。この操作は元に戻せません。続けますか？')){$('bodyText').value='';updateAndSave();$('bodyText').focus()}});
$('resetBtn').addEventListener('click',()=>{if(confirm('用紙設定を初期値に戻しますか？本文は消去されません。')){const body=$('bodyText').value;settings={...DEFAULTS,bodyText:body};$('rows').value=DEFAULTS.rows;$('cols').value=DEFAULTS.cols;$('cellSize').value=DEFAULTS.cellSize;$('gridColor').value=DEFAULTS.gridColor;$('font').value=DEFAULTS.font;$('fontSize').value=DEFAULTS.fontSize;$('title').value='';settings.direction='horizontal';updateAndSave()}});
window.addEventListener('beforeprint',()=>{render()});restore();render();
